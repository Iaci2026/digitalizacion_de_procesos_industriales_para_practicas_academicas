        function post(url, obj) {

            var x = new XMLHttpRequest();

            x.open('POST', url, true);

            x.setRequestHeader(
                'Content-Type',
                'application/json'
            );

            x.send(JSON.stringify(obj));

        }

function update() {

    function f2(v) {
    return (v == null) ? "--" : Number(v).toFixed(2);
    }
    var x = new XMLHttpRequest();

    x.open('GET', '/datos', true);

    x.timeout = 3000;

    x.onreadystatechange = function () {

        if (x.readyState == 4 && x.status == 200) {

            try {

                var d = JSON.parse(x.responseText);

                //
                // VARIABLES
                //

                document.getElementById('Temperatura_html').innerHTML =
                    d.Temperatura_html ?? "--";

                document.getElementById('Temperatura_html2').innerHTML =
                    d.Temperatura_html ?? "--";

                document.getElementById('Nivel_html').innerHTML =
                    d.Nivel_html ?? "--";

                document.getElementById('Nivel_html2').innerHTML =
                    d.Nivel_html ?? "--";

                document.getElementById('PresionAgua_html').innerHTML =
                    d.PresionAgua_html ?? "--";

                document.getElementById('PresionAire_html').innerHTML =
                    d.PresionAire_html ?? "--";

                document.getElementById('Caudal_html').innerHTML =
                    d.Caudal_html ?? "--";

                document.getElementById('SP_Temp_html').innerHTML =
                    d.SP_Temp_html ?? "--";

                document.getElementById('SP_Niv_html').innerHTML =
                    d.SP_Niv_html ?? "--";

                document.getElementById('Kp_Nivel_html').innerHTML = f2(d.Kp_Nivel_html);
                document.getElementById('Ki_Nivel_html').innerHTML = f2(d.Ki_Nivel_html);
                document.getElementById('Kd_Nivel_html').innerHTML = f2(d.Kd_Nivel_html);
                
                document.getElementById('Kp_Temp_html').innerHTML = f2(d.Kp_Temp_html);
                document.getElementById('Ki_Temp_html').innerHTML = f2(d.Ki_Temp_html);
                document.getElementById('Kd_Temp_html').innerHTML = f2(d.Kd_Temp_html);

                //
                // SALIDAS PID
                //

                document.getElementById('Salida_PID_temperatura').innerHTML =
                    d.Salida_PID_temperatura ?? "--";

                document.getElementById('Salida_PID_Nivel').innerHTML =
                    d.Salida_PID_Nivel ?? "--";

                //
                // LEDS
                //

                setLed("Confirmacion_km1_led", d.ConfirmacionKM1);


                setLed("Luz_mb_001_led", d.MB001);

                setLed("Luz_mb_002_led", d.MB002);

                setLed("Luz_nivel", d.Eleccion_proceso_html == 0);

                setLed("Luz_temperatura", d.Eleccion_proceso_html);

                setLed("Luz_pid_off_niv", d.PIDNivelOFF);

                setLed("Luz_pid_off_temp", d.PIDTempOFF);

                setLed("Luz_pid_auto_temp", d.Autotemp);

                setLed("Luz_pid_auto_niv", d.Autoniv);

                setLed("Luz_cool_led", d.Cooler);

                setLed("Luz_res_led", d.Resistencia);

                setLed("Luz_reset_alarmas", d.SendResetAlarm);

                //
                // ALARMAS
                //
                document.getElementById('Confirmacion_HTML_led').innerHTML =
                d.Confirmacion_HTML_led ? "DESCONECTADO" : "CONECTADO";

                document.getElementById('AlarmaParo_html').innerHTML =
                d.AlarmaParo_html ? "ACTIVA" : "NORMAL";            


                document.getElementById('alarma_nivel_alto_alto').innerHTML =
                    d.AlarmaNivelAA_html ? "ACTIVA" : "NORMAL";

                document.getElementById('alarma_temperatura_alta').innerHTML =
                    d.AlarmaTempAlta_html ? "ACTIVA" : "NORMAL";

                document.getElementById('alarma_aire_bajo').innerHTML =
                    d.AlarmaPresionAireBaja_html ? "ACTIVA" : "NORMAL";

                document.getElementById('alarma_nivel_alto').innerHTML =
                    d.AlarmaNivelAlto_html ? "ACTIVA" : "NORMAL";

            } catch (e) {

                console.log("ERROR JSON:", e);

            }

        }

    };

    x.onerror = function () {

        console.log("ERROR AJAX");

    };

    x.ontimeout = function () {

        console.log("TIMEOUT");

    };

    x.send();

}

        setInterval(update, 2000);

        update();
